# backend/services/ai_service.py (lines 87-175)
async def generate_completion(
    prompt: str,
    system_prompt: Optional[str] = None,
    provider: Literal['claude', 'gemini'] = 'claude',
    max_tokens: int = 4096,
    temperature: float = 0.7,
    user_id: Optional[str] = None,
    user_email: Optional[str] = None,
    service_name: str = "unknown",
    service_action: str = "generate"
) -> AIResponse:
    """
    Unified AI completion function that routes to any provider via LiteLLM.
    Automatically logs usage to the database.
    """
    model = PROVIDER_MODELS.get(provider)
    
    messages = []
    if system_prompt:
        messages.append({"role": "system", "content": system_prompt})
    messages.append({"role": "user", "content": prompt})
    
    start_time = time.time()
    
    # Call LiteLLM (routes to correct provider)
    response = await litellm.acompletion(
        model=model,
        messages=messages,
        max_tokens=max_tokens,
        temperature=temperature
    )
    
    # Extract usage data
    content = response.choices[0].message.content
    input_tokens = response.usage.prompt_tokens
    output_tokens = response.usage.completion_tokens
    total_tokens = response.usage.total_tokens
    
    # Calculate cost
    cost = litellm.completion_cost(completion_response=response)
    
    # Log usage to database
    await log_ai_usage(...)
    
    return AIResponse(
        content=content,
        input_tokens=input_tokens,
        output_tokens=output_tokens,
        total_tokens=total_tokens,
        model=model,
        provider=provider,
        cost=cost
    )

def build_smart_questions_prompt(job_data: str, cv_text: Optional[str] = None):
    prompt = """CRITICAL: You must respond with ONLY valid JSON. 
    No text before or after. No markdown. No code blocks.
    You are an INTELLIGENT HUMAN INTERVIEWER who has carefully studied 
    both the job description AND the candidate's background...
    
    JOB DESCRIPTION:
    """ + job_data[:4000]
    
    if cv_text:
        prompt += """
        CANDIDATE'S CV/BACKGROUND:
        """ + cv_text[:2000]
    
    prompt += """
    Generate a JSON response with this EXACT structure:
    {"focus_areas":[...], "questions":[...]}
    """
    return prompt

# backend/app/smart_questions.py
def parse_gemini_response(response_text: str) -> dict:
    text = response_text.strip()
    
    # Remove markdown code blocks
    if text.startswith("```json"):
        text = text[7:]
    if text.endswith("```"):
        text = text[:-3]
    
    # Find JSON boundaries
    start_idx = text.find('{')
    end_idx = text.rfind('}')
    text = text[start_idx:end_idx + 1]
    
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        # Recovery: attempt to fix truncated JSON
        for i in range(len(text), 0, -50):
            test_text = text[:i]
            # Close unclosed brackets/braces
            open_brackets = test_text.count('[') - test_text.count(']')
            open_braces = test_text.count('{') - test_text.count('}')
            test_text += ']' * max(0, open_brackets)
            test_text += '}' * max(0, open_braces)
            try:
                return json.loads(test_text)
            except:
                continue
        return {"error": "Failed to parse response"}
Error Handling
# backend/services/ai_service.py
try:
    response = await litellm.acompletion(...)
    
except Exception as e:
    duration_ms = int((time.time() - start_time) * 1000)
    logger.error(f"AI generation failed with {provider}: {str(e)}")
    
    # Log failure to database
    await log_ai_usage(
        user_id=user_id,
        service_name=service_name,
        provider=provider,
        model=model,
        input_tokens=0,
        output_tokens=0,
        total_tokens=0,
        cost_usd=None,
        duration_ms=duration_ms,
        success=False,
        error_message=str(e),
        user_email=user_email,
        service_action=service_action
    )
    raise  # Re-raise exception

Usage Logging Function
# backend/services/ai_service.py (lines 32-70)
async def log_ai_usage(
    user_id: Optional[str],
    service_name: str,
    provider: str,
    model: str,
    input_tokens: int,
    output_tokens: int,
    total_tokens: int,
    cost_usd: Optional[float],
    duration_ms: int,
    success: bool,
    error_message: Optional[str] = None,
    user_email: Optional[str] = None,
    service_action: Optional[str] = None
):
    """Log AI usage to the database for tracking and billing."""
    cursor.execute(
        """INSERT INTO ai_usage_logs 
           (user_id, service_name, provider, model, input_tokens, output_tokens, 
            total_tokens, cost_usd, duration_ms, success, error_message, 
            user_email, service_action)
           VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)""",
        (user_id, service_name, provider, model, input_tokens, output_tokens,
         total_tokens, float(cost_usd) if cost_usd else None, duration_ms, 
         success, error_message, user_email, service_action)
    )

Database Schema for Tracking
-- Table: ai_usage_logs
CREATE TABLE ai_usage_logs (
    id UUID PRIMARY KEY,
    user_id UUID REFERENCES users(id),
    service_name VARCHAR(50) NOT NULL,     -- 'xray', 'smart_questions', 'cv_optimizer'
    provider VARCHAR(20) NOT NULL,          -- 'claude', 'gemini'
    model VARCHAR(100) NOT NULL,            -- Full model name
    input_tokens INTEGER,
    output_tokens INTEGER,
    total_tokens INTEGER,
    cost_usd DECIMAL(10, 6),
    duration_ms INTEGER,
    success BOOLEAN DEFAULT TRUE,
    error_message TEXT,
    user_email VARCHAR(255),
    service_action VARCHAR(50),             -- 'generate', 'analyze', 'fix'
    created_at TIMESTAMPTZ DEFAULT NOW()
);
