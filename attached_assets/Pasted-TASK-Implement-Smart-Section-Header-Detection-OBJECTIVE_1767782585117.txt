def _is_section_header_text(text: str) -> bool:
    """
    Check if text is a CV section header based on keywords.
    
    Returns True for: "About", "Experience", "Skills"
    Returns False for: "Citi", "Microsoft", "Core Expertise & Impact"
    """
    if not text or len(text) > 50:
        return False
    
    # Normalize: lowercase, strip whitespace
    normalized = text.lower().strip()
    # Remove special characters but keep spaces
    normalized = re.sub(r'[^a-z\s]', '', normalized).strip()
    
    # Direct match
    if normalized in SECTION_HEADER_KEYWORDS:
        return True
    
    # Check if starts with keyword (e.g., "Skills & Expertise")
    for keyword in SECTION_HEADER_KEYWORDS:
        if normalized.startswith(keyword + ' '):
            return True
    
    return False
